59: export function isRequestDeadlineMissed(
60:   item?: Pick<PublicRecordItem, 'dueDate' | 'status'> | null,
61:   todayIso?: string
62: ): boolean {
63:   if (!item || !item.dueDate) return false;
64:   const status = (item.status || '').toLowerCase();
65:   if (status === 'fulfilled' || status === 'closed' || status === 'completed') return false;
66:   const dueMatch = /^(\d{4}-\d{2}-\d{2})/.exec(item.dueDate.trim());
67:   if (!dueMatch) return false;
68:   const refDate = todayIso && /^\d{4}-\d{2}-\d{2}/.test(todayIso.trim())
69:     ? todayIso.trim().slice(0, 10)
70:     : new Date().toISOString().slice(0, 10);
71:   return dueMatch[1] < refDate;
72: }

188: function cleanSubjectTitle(rawTitle: string, statuteCode: string): string {
189:   const statuteWord = statuteCode.trim();

366: export function buildMailtoUrl(payload: HandoffPayload): string {
367:   const recipient = encodeURIComponent(normalizeRecipientEmail(payload.to));
368:   const queryParts: string[] = [];
369:   if (payload.subject.trim()) {
370:     queryParts.push(`subject=${encodeURIComponent(payload.subject.trim())}`);
371:   }
372:   if (payload.body.trim()) {
373:     queryParts.push(`body=${encodeURIComponent(payload.body)}`);
374:   }
375:   return `mailto:${recipient}${queryParts.length > 0 ? `?${queryParts.join('&')}` : ''}`;
